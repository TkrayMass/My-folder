CREATE OR REPLACE PROCEDURE MHTEAM.MHA.SNAPSHOT_MONTHLY_DB()

RETURNS VARCHAR
LANGUAGE JAVASCRIPT
EXECUTE AS CALLER
AS
$$
try {

  var rs = snowflake.execute({
    sqlText: `
      SELECT
        LAST_DAY(DATEADD(MONTH,-1, CURRENT_DATE()))::DATE AS SNAPDT,
        DATEADD(DAY, 1, LAST_DAY(DATEADD(MONTH,-1, CURRENT_DATE())))::DATE AS EXTRDT,
        TO_NUMBER(TO_CHAR(LAST_DAY(DATEADD(MONTH,-1, CURRENT_DATE())), 'YYYYMM')) AS YR_MTH
    `
  });
  rs.next();

  // Convert JS Date → YYYY-MM-DD string
  function toYMD(d) {
    return d.toISOString().split('T')[0];
  }

  var snapdt = toYMD(rs.getColumnValue("SNAPDT"));
  var extrdt = toYMD(rs.getColumnValue("EXTRDT"));
  var yr_mth = rs.getColumnValue("YR_MTH");

  snowflake.execute({
    sqlText: `
      CREATE OR REPLACE TEMP TABLE TMP_SNAPSHOT AS
      SELECT
        ${yr_mth} AS YR_MTH,
        LPAD(TRIM(COALESCE(se.cde_budget_group,'00')), 2, '0') AS CDE_BUDGET_GROUP,
        COUNT(*) AS MEMBERS
      FROM MHDWPROD.NW.NW_STATE_ELIGIBILITY_HIST se
      WHERE TO_DATE('${extrdt}') BETWEEN TO_DATE(se.valid_from_dt_tm) AND TO_DATE(se.valid_thru_dt_tm)
        AND se.ind_active = 'Y'
        AND TO_DATE('${snapdt}') >= se.dte_effective
        AND TO_DATE('${snapdt}') < DATEADD(DAY, 1, se.dte_end)
      GROUP BY 1,2
    `
  });

  snowflake.execute({
    sqlText: `
      DELETE FROM MHTEAM.MHA.BUDGET_MONTHLY_SNAPSHOT_DB
      WHERE YR_MTH = ${yr_mth}
    `
  });

  snowflake.execute({
    sqlText: `
      INSERT INTO MHTEAM.MHA.BUDGET_MONTHLY_SNAPSHOT_DB (
        YR_MTH,
        CDE_BUDGET_GROUP,
        MEMBERS
      )
      SELECT
        YR_MTH,
        CDE_BUDGET_GROUP,
        MEMBERS
      FROM TMP_SNAPSHOT
    `
  });

  var qc = snowflake.execute({
    sqlText: `
      SELECT COUNT(*) AS ROWS_LOADED, COALESCE(SUM(MEMBERS),0) AS TOTAL_MEMBERS
      FROM MHTEAM.MHA.BUDGET_MONTHLY_SNAPSHOT_DB
      WHERE YR_MTH = ${yr_mth}
    `
  });
  qc.next();

  return 'Snapshot loaded for ' + yr_mth +
         ', rows=' + qc.getColumnValue("ROWS_LOADED") +
         ', total=' + qc.getColumnValue("TOTAL_MEMBERS");

} catch (err) {
  return 'Error: ' + err.message;
}
$$;

  return '✅ Snapshot loaded for ' + yr_mth;

} catch (err) {
  return '❌ Error: ' + err.message;
}
$$;
