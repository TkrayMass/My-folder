
CREATE OR REPLACE PROCEDURE MHTEAM.MHA.SNAPSHOT_202604_0430()
RETURNS STRING
LANGUAGE JAVASCRIPT
EXECUTE AS CALLER
AS
$$
try {

  var yr_mth = '202604';
  var snapdt = '2026-04-30';
  var extrdt = '2026-05-01';

  snowflake.execute({
    sqlText: `
      CREATE OR REPLACE TABLE MHTEAM.MHA.SNAPSHOT_202604_0430 AS
      SELECT
        ${yr_mth} AS YR_MTH,
        COALESCE(se.cde_budget_group, '00') AS CDE_BUDGET_GROUP,
        COUNT(DISTINCT se.id_medicaid) AS MEMBERS
      FROM MHDWPROD.NW.NW_STATE_ELIGIBILITY_HIST se
      WHERE TO_DATE('${extrdt}') BETWEEN TO_DATE(se.valid_from_dt_tm)
                                      AND TO_DATE(se.valid_thru_dt_tm)
        AND se.ind_active = 'Y'
        AND TO_DATE('${snapdt}') >= se.dte_effective
        AND TO_DATE('${snapdt}') < DATEADD(DAY, 1, se.dte_end)
      GROUP BY
        COALESCE(se.cde_budget_group, '00')
      ORDER BY
        CDE_BUDGET_GROUP
    `
  });

  var qc = snowflake.execute({
    sqlText: `
      SELECT
        COUNT(*) AS ROWS_LOADED,
        COALESCE(SUM(MEMBERS),0) AS TOTAL_MEMBERS,
        COALESCE(SUM(CASE WHEN CDE_BUDGET_GROUP NOT IN ('44','87','99')
                          THEN MEMBERS ELSE 0 END),0) AS TOTAL_EXCL_44_87_99
      FROM MHTEAM.MHA.SNAPSHOT_202604_0430
    `
  });

  qc.next();

  return 'Snapshot 202604 completed using DISTINCT members. Rows='
       + qc.getColumnValue('ROWS_LOADED')
       + ', Total='
       + qc.getColumnValue('TOTAL_MEMBERS')
       + ', Total excluding 44/87/99='
       + qc.getColumnValue('TOTAL_EXCL_44_87_99');

} catch (err) {
  return 'Error: ' + err.message;
}
$$;


CALL MHTEAM.MHA.SNAPSHOT_202604_0430();

Select * from MHTEAM.MHA.SNAPSHOT_202604_0430;
