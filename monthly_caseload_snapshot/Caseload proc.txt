CREATE OR REPLACE PROCEDURE MHTEAM.MHA.CASELOAD_MONTHLY_DB2(
  "START_MONTH" DATE DEFAULT '2006-07-01',
  "END_MONTH"   DATE DEFAULT CURRENT_DATE()
)
RETURNS VARCHAR
LANGUAGE JAVASCRIPT
EXECUTE AS CALLER
AS
$$
/*============================================================
  CASELOAD_MONTHLY_DB2  (SAS-equivalent monthly member-days)

  PURPOSE
  - Rebuild monthly caseload member-days by budget group.
  - SAS-equivalent logic:
      * ASOFDT = first day of the month the job runs
      * TRUE_END = last day of the prior month relative to ASOFDT
        (also respects END_MONTH month-end)
      * Validity window slice uses ASOFDT (NOT CURRENT_DATE)

  INPUTS
  - START_MONTH: earliest month to compute (normalized to 1st of month)
  - END_MONTH  : user-provided end; capped by TRUE_END

  OUTPUTS (overwritten each run)
  - MHTEAM.MHA.CASELOAD_MEMBER_DAYS_TEMP (working)
  - MHTEAM.MHA.CASELOAD_MEMBER_DAYS      (final)

  DUPLICATE SAFETY
  - Final output table is rebuilt with CREATE OR REPLACE, so reruns
    do not duplicate rows.

  NOTES
  - Uses timestamp-safe validity slicing:
      ASOF_TS >= VALID_FROM_DT_TM AND ASOF_TS < VALID_THRU_DT_TM
  - Budget group standardized to 2 chars with LPAD + default '00'
============================================================*/

function ymdUTC(d) {
  // Snowflake JS Date -> YYYY-MM-DD string in UTC (prevents timezone drift)
  var y  = d.getUTCFullYear();
  var m  = String(d.getUTCMonth() + 1).padStart(2,'0');
  var da = String(d.getUTCDate()).padStart(2,'0');
  return `${y}-${m}-${da}`;
}

var months_inserted = 0;

//------------------------------------------------------------
// 1) ASOFDT = first day of run month (SAS logic)
//------------------------------------------------------------
var asof_stmt = snowflake.execute({
  sqlText: "SELECT DATE_TRUNC('MONTH', CURRENT_DATE())::DATE"
});
asof_stmt.next();
var as_of_dt = asof_stmt.getColumnValue(1);     // JS Date
var asof_str = ymdUTC(as_of_dt);                // 'YYYY-MM-DD'
var asof_ts  = `${asof_str} 00:00:00`;          // midnight timestamp string

//------------------------------------------------------------
// 2) Normalize START_MONTH to first of its month
//------------------------------------------------------------
var start_stmt = snowflake.execute({
  sqlText: "SELECT DATE_TRUNC('MONTH', TO_DATE(?,'YYYY-MM-DD'))::DATE",
  binds: [ ymdUTC(START_MONTH) ]
});
start_stmt.next();
var begm = start_stmt.getColumnValue(1);        // JS Date (month start)

//------------------------------------------------------------
// 3) TRUE_END = last day of prior month relative to ASOFDT,
//    but also respect END_MONTH month end.
//------------------------------------------------------------
var true_end_stmt = snowflake.execute({
  sqlText: `
    SELECT LEAST(
             LAST_DAY(DATEADD(month, -1, TO_DATE(?,'YYYY-MM-DD')))::DATE,
             LAST_DAY(TO_DATE(?,'YYYY-MM-DD'))::DATE
           )
  `,
  binds: [ asof_str, ymdUTC(END_MONTH) ]
});
true_end_stmt.next();
var true_end = true_end_stmt.getColumnValue(1); // JS Date

//------------------------------------------------------------
// 4) Create working table (overwrite each run)
//------------------------------------------------------------
snowflake.execute({
  sqlText: `
    CREATE OR REPLACE TABLE MHTEAM.MHA.CASELOAD_MEMBER_DAYS_TEMP (
      YR_MTH           NUMBER,
      CDE_BUDGET_GROUP STRING,
      MEM_DAYS         NUMBER
    )
  `
});

//------------------------------------------------------------
// 5) Loop months from START_MONTH .. TRUE_END
//------------------------------------------------------------
while (begm <= true_end) {

  var begm_str = ymdUTC(begm); // month start YYYY-MM-DD

  // endm = last day of begm month (capped at true_end)
  var endm_stmt = snowflake.execute({
    sqlText: "SELECT LAST_DAY(TO_DATE(?,'YYYY-MM-DD'))::DATE",
    binds: [ begm_str ]
  });
  endm_stmt.next();
  var endm = endm_stmt.getColumnValue(1); // JS Date

  if (endm > true_end) endm = true_end;
  var endm_str = ymdUTC(endm);

  // yr_mth = YYYYMM numeric
  var ym_stmt = snowflake.execute({
    sqlText: "SELECT TO_NUMBER(TO_CHAR(TO_DATE(?,'YYYY-MM-DD'),'YYYYMM'))",
    binds: [ begm_str ]
  });
  ym_stmt.next();
  var yr_mth = ym_stmt.getColumnValue(1);

  //----------------------------------------------------------
  // Insert month slice (SAS-equivalent overlap logic)
  // - Valid slice uses ASOFDT timestamp (boundary-safe)
  // - Overlap arithmetic uses DATEDIFF + 1, matching SAS logic
  // - Budget group standardized to 2 chars, default '00'
  //----------------------------------------------------------
  var insert_sql = `
    INSERT INTO MHTEAM.MHA.CASELOAD_MEMBER_DAYS_TEMP
    SELECT
      ${yr_mth} AS yr_mth,
      LPAD(TRIM(NVL(se.cde_budget_group, '00')), 2, '0') AS cde_budget_group,
      SUM(
        CASE
          WHEN TO_DATE(se.dte_effective) < TO_DATE('${begm_str}')
           AND TO_DATE(se.dte_end)       > TO_DATE('${endm_str}')
            THEN DATEDIFF(day, TO_DATE('${begm_str}'), TO_DATE('${endm_str}')) + 1

          WHEN TO_DATE(se.dte_effective) < TO_DATE('${begm_str}')
            THEN DATEDIFF(day, TO_DATE('${begm_str}'), TO_DATE(se.dte_end)) + 1

          WHEN TO_DATE(se.dte_end) > TO_DATE('${endm_str}')
            THEN DATEDIFF(day, TO_DATE(se.dte_effective), TO_DATE('${endm_str}')) + 1

          ELSE DATEDIFF(day, TO_DATE(se.dte_effective), TO_DATE(se.dte_end)) + 1
        END
      ) AS mem_days
    FROM MHDWPROD.NW.NW_STATE_ELIGIBILITY_HIST se
    WHERE TO_TIMESTAMP_NTZ('${asof_ts}') >= se.valid_from_dt_tm
      AND TO_TIMESTAMP_NTZ('${asof_ts}') <  se.valid_thru_dt_tm
      AND se.ind_active = 'Y'
      AND TO_DATE(se.dte_effective) <= TO_DATE('${endm_str}')
      AND TO_DATE(se.dte_end)       >= TO_DATE('${begm_str}')
    GROUP BY LPAD(TRIM(NVL(se.cde_budget_group, '00')), 2, '0')
  `;

  snowflake.execute({ sqlText: insert_sql });
  months_inserted++;

  // next month
  var nxt_stmt = snowflake.execute({
    sqlText: "SELECT DATEADD(month, 1, TO_DATE(?,'YYYY-MM-DD'))::DATE",
    binds: [ begm_str ]
  });
  nxt_stmt.next();
  begm = nxt_stmt.getColumnValue(1);
}

//------------------------------------------------------------
// 6) Final output table (overwritten each run)
//------------------------------------------------------------
snowflake.execute({
  sqlText: `
    CREATE OR REPLACE TABLE MHTEAM.MHA.CASELOAD_MEMBER_DAYS AS
    SELECT *
    FROM MHTEAM.MHA.CASELOAD_MEMBER_DAYS_TEMP
    ORDER BY YR_MTH ASC, CDE_BUDGET_GROUP ASC
  `
});

return "✅ CASELOAD_MEMBER_DAYS completed through " + ymdUTC(true_end) +
       " | ASOFDT=" + asof_str +
       " | months_inserted=" + months_inserted;

$$;
