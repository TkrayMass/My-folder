CREATE OR REPLACE PROCEDURE MHTEAM.MHA.SNAPSHOT_MONTHLY_DYNAMIC()
RETURNS STRING
LANGUAGE JAVASCRIPT
EXECUTE AS CALLER
AS
$$
try {

  snowflake.execute({
    sqlText: `
      CREATE OR REPLACE TABLE MHTEAM.MHA.SNAPSHOT_MONTHLY_DYNAMIC AS
      WITH dates AS (
        SELECT
          LAST_DAY(DATEADD(MONTH, -1, CURRENT_DATE()))::DATE AS SNAPDT,
          DATE_TRUNC('MONTH', CURRENT_DATE())::DATE AS EXTRDT,
          TO_NUMBER(TO_CHAR(LAST_DAY(DATEADD(MONTH, -1, CURRENT_DATE())), 'YYYYMM')) AS YR_MTH
      )
      SELECT
        d.YR_MTH,
        LPAD(TRIM(COALESCE(se.cde_budget_group, '00')), 2, '0') AS CDE_BUDGET_GROUP,
        COUNT(DISTINCT se.id_medicaid) AS MEMBERS
      FROM MHDWPROD.NW.NW_STATE_ELIGIBILITY_HIST se
      CROSS JOIN dates d
      WHERE TO_TIMESTAMP_NTZ(d.EXTRDT) >= se.valid_from_dt_tm
        AND TO_TIMESTAMP_NTZ(d.EXTRDT) <  se.valid_thru_dt_tm
        AND se.ind_active = 'Y'
        AND d.SNAPDT BETWEEN TO_DATE(se.dte_effective)
                         AND TO_DATE(se.dte_end)
      GROUP BY
        d.YR_MTH,
        LPAD(TRIM(COALESCE(se.cde_budget_group, '00')), 2, '0')
      ORDER BY
        CDE_BUDGET_GROUP
    `
  });

  var qc = snowflake.execute({
    sqlText: `
      SELECT
        MIN(YR_MTH) AS YR_MTH,
        COUNT(*) AS ROWS_LOADED,
        COALESCE(SUM(MEMBERS),0) AS TOTAL_MEMBERS,
        COALESCE(SUM(CASE WHEN CDE_BUDGET_GROUP NOT IN ('44','87','99')
                          THEN MEMBERS ELSE 0 END),0) AS TOTAL_EXCL_44_87_99
      FROM MHTEAM.MHA.SNAPSHOT_MONTHLY_DYNAMIC
    `
  });

  qc.next();

  return 'Snapshot completed. YR_MTH='
       + qc.getColumnValue('YR_MTH')
       + ', Rows='
       + qc.getColumnValue('ROWS_LOADED')
       + ', Total='
       + qc.getColumnValue('TOTAL_MEMBERS')
       + ', Total excluding 44/87/99='
       + qc.getColumnValue('TOTAL_EXCL_44_87_99');

} catch (err) {
  return 'Error: ' + err.message;
}
$$;



CALL MHTEAM.MHA.SNAPSHOT_MONTHLY_DYNAMIC();

SELECT *
FROM MHTEAM.MHA.SNAPSHOT_MONTHLY_DYNAMIC
ORDER BY YR_MTH, CDE_BUDGET_GROUP;










CREATE OR REPLACE PROCEDURE MHTEAM.MHA.CASELOAD_MONTHLY_DB2(
  START_MONTH DATE DEFAULT '2006-07-01',
  END_MONTH   DATE DEFAULT CURRENT_DATE()
)
RETURNS VARCHAR
LANGUAGE JAVASCRIPT
EXECUTE AS CALLER
AS
$$
function ymdUTC(d) {
  var y  = d.getUTCFullYear();
  var m  = String(d.getUTCMonth() + 1).padStart(2,'0');
  var da = String(d.getUTCDate()).padStart(2,'0');
  return `${y}-${m}-${da}`;
}

var months_inserted = 0;

/* ASOFDT = first day of current run month */
var asof_stmt = snowflake.execute({
  sqlText: "SELECT DATE_TRUNC('MONTH', CURRENT_DATE())::DATE"
});
asof_stmt.next();

var as_of_dt = asof_stmt.getColumnValue(1);
var asof_str = ymdUTC(as_of_dt);
var asof_ts  = `${asof_str} 00:00:00`;

/* START_MONTH normalized to first of month */
var start_stmt = snowflake.execute({
  sqlText: "SELECT DATE_TRUNC('MONTH', TO_DATE(?,'YYYY-MM-DD'))::DATE",
  binds: [ymdUTC(START_MONTH)]
});
start_stmt.next();

var begm = start_stmt.getColumnValue(1);

/* TRUE_END = last day of prior month, capped by END_MONTH */
var true_end_stmt = snowflake.execute({
  sqlText: `
    SELECT LEAST(
      LAST_DAY(DATEADD(MONTH, -1, TO_DATE(?,'YYYY-MM-DD')))::DATE,
      LAST_DAY(TO_DATE(?,'YYYY-MM-DD'))::DATE
    )
  `,
  binds: [asof_str, ymdUTC(END_MONTH)]
});
true_end_stmt.next();

var true_end = true_end_stmt.getColumnValue(1);

/* Working table */
snowflake.execute({
  sqlText: `
    CREATE OR REPLACE TABLE MHTEAM.MHA.CASELOAD_MEMBER_DAYS_TEMP (
      YR_MTH           NUMBER,
      CDE_BUDGET_GROUP STRING,
      MEM_DAYS         NUMBER
    )
  `
});

/* Loop months */
while (begm <= true_end) {

  var begm_str = ymdUTC(begm);

  var endm_stmt = snowflake.execute({
    sqlText: "SELECT LAST_DAY(TO_DATE(?,'YYYY-MM-DD'))::DATE",
    binds: [begm_str]
  });
  endm_stmt.next();

  var endm = endm_stmt.getColumnValue(1);
  if (endm > true_end) endm = true_end;

  var endm_str = ymdUTC(endm);

  var ym_stmt = snowflake.execute({
    sqlText: "SELECT TO_NUMBER(TO_CHAR(TO_DATE(?,'YYYY-MM-DD'),'YYYYMM'))",
    binds: [begm_str]
  });
  ym_stmt.next();

  var yr_mth = ym_stmt.getColumnValue(1);

  var insert_sql = `
    INSERT INTO MHTEAM.MHA.CASELOAD_MEMBER_DAYS_TEMP
    SELECT
      ${yr_mth} AS YR_MTH,
      LPAD(TRIM(NVL(se.cde_budget_group, '00')), 2, '0') AS CDE_BUDGET_GROUP,
      SUM(
        DATEDIFF(
          DAY,
          GREATEST(TO_DATE(se.dte_effective), TO_DATE('${begm_str}')),
          LEAST(TO_DATE(NVL(se.dte_end, '9999-12-31')), TO_DATE('${endm_str}'))
        ) + 1
      ) AS MEM_DAYS
    FROM MHDWPROD.NW.NW_STATE_ELIGIBILITY_HIST se
    WHERE TO_TIMESTAMP_NTZ('${asof_ts}') >= se.valid_from_dt_tm
      AND TO_TIMESTAMP_NTZ('${asof_ts}') <  se.valid_thru_dt_tm
      AND se.ind_active = 'Y'
      AND TO_DATE(se.dte_effective) <= TO_DATE('${endm_str}')
      AND TO_DATE(NVL(se.dte_end, '9999-12-31')) >= TO_DATE('${begm_str}')
    GROUP BY
      LPAD(TRIM(NVL(se.cde_budget_group, '00')), 2, '0')
  `;

  snowflake.execute({ sqlText: insert_sql });
  months_inserted++;

  var nxt_stmt = snowflake.execute({
    sqlText: "SELECT DATEADD(MONTH, 1, TO_DATE(?,'YYYY-MM-DD'))::DATE",
    binds: [begm_str]
  });
  nxt_stmt.next();

  begm = nxt_stmt.getColumnValue(1);
}

/* Final output */
snowflake.execute({
  sqlText: `
    CREATE OR REPLACE TABLE MHTEAM.MHA.CASELOAD_MEMBER_DAYS AS
    SELECT
      YR_MTH,
      CDE_BUDGET_GROUP,
      MEM_DAYS,
      TO_DATE('${asof_str}') AS ASOFDT
    FROM MHTEAM.MHA.CASELOAD_MEMBER_DAYS_TEMP
    ORDER BY YR_MTH, CDE_BUDGET_GROUP
  `
});

return 'CASELOAD_MEMBER_DAYS completed through ' + ymdUTC(true_end) +
       ' | ASOFDT=' + asof_str +
       ' | months_inserted=' + months_inserted;
$$;


CALL MHTEAM.MHA.CASELOAD_MONTHLY_DB2();

SELECT
    YR_MTH,
    CDE_BUDGET_GROUP,
    MEM_DAYS
FROM MHTEAM.MHA.CASELOAD_MEMBER_DAYS
ORDER BY
    YR_MTH,
    CDE_BUDGET_GROUP;
