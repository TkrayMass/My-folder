USE ROLE MHA_TEAM_ROLE;
USE DATABASE MHDWPROD;
USE SCHEMA MHA;

CREATE OR REPLACE PROCEDURE ACOENR_WEEKLY_REFRESH()
RETURNS VARCHAR
LANGUAGE SQL
EXECUTE AS OWNER
AS
$$
DECLARE
    StartDate  DATE DEFAULT TO_DATE('2023-03-01');
    today      DATE DEFAULT CURRENT_DATE();
    EndDate    DATE DEFAULT CURRENT_DATE();
    lastweek   DATE DEFAULT DATEADD(DAY, -7, CURRENT_DATE());
    wh_thru_dt DATE DEFAULT CURRENT_DATE();
    repdt      DATE DEFAULT CURRENT_DATE();

BEGIN

    DROP TABLE IF EXISTS weektable_;

    CREATE OR REPLACE TABLE weektable_ AS
    SELECT    
        'Month of' AS Timeperiod,
        calendar_year,
        calendar_week_num, 
        dt,     
        num_month,     
        dt AS startdt, 
        dt AS enddt, 
        dt_yyyymmdd AS firstdtweek
    FROM MHDWPROD.NW.nw_date
    WHERE calendar_year >= 2023 
      AND dt >= :StartDate 
      AND dt <= :EndDate

    UNION ALL

    SELECT  
        'Day of' AS Timeperiod,
        calendar_year, 
        dt_yyyymmdd AS calendar_week_num, 
        dt, 
        MIN(num_month) AS num_month, 
        MIN(dt) AS startdt, 
        MAX(dt) AS enddt, 
        MIN(dt_yyyymmdd) AS firstdtweek
    FROM MHDWPROD.NW.nw_date
    WHERE calendar_year >= 2023 
      AND dt > :lastweek 
      AND dt <= :today
    GROUP BY calendar_year, dt_yyyymmdd, dt
    ORDER BY calendar_year, calendar_week_num, num_month, dt;


    CREATE OR REPLACE TABLE test_weektable2_ AS 
    SELECT 
        calendar_year,
        'Month of' AS Timeperiod, 
        num_month,
        MIN(dt) AS startdt, 
        MAX(dt) AS enddt, 
        MIN(dt_yyyymmdd) AS firstdtweek
    FROM MHDWPROD.NW.nw_date
    WHERE calendar_year >= 2023 
      AND dt >= :StartDate   
      AND dt <= :EndDate     
    GROUP BY calendar_year, num_month;


    CREATE OR REPLACE TABLE nw_member_enroll AS
    SELECT mem.*
    FROM MHDWPROD.NW.nw_member mem
    WHERE mem.id_medicaid NOT IN (
      '100231397561',
      '100232588994',
      '100232637148',
      '100019999638',
      '100024398297',
      '100220572927',
      '100008695023'
    );


    CREATE OR REPLACE TABLE a_Region_SrvArea AS
    SELECT  
      mem.sak_recip, 
      mem.mem_seq, 
      mem.id_medicaid, 
      addr.ADR_ZIP_CODE AS RES_ZIP,
      MCO_REGION, 
      SERVICE_AREA, 
      CDE_MC_REGION_CAP, 
      DSC_MC_REGION_CAP, 
      dsc_mc_region_enroll, 
      COUNTY
    FROM nw_member_enroll mem
    INNER JOIN (
        SELECT sak_recip, cde_addr_usage, ADR_ZIP_CODE, valid_from_dt_tm, valid_thru_dt_tm 
        FROM MHDWPROD.NW.nw_member_addr_hist
    ) addr
      ON addr.sak_recip = mem.sak_recip 
      AND addr.cde_addr_usage = 'MR' 
      AND :today BETWEEN addr.valid_from_dt_tm AND addr.valid_thru_dt_tm
    INNER JOIN (
        SELECT ADR_MAIL_ZIP, SERVICE_AREA, CDE_MC_REGION_CAP, DSC_MC_REGION_CAP, 
               dsc_mc_region_enroll, ADR_MAIL_COUNTY_DSP AS COUNTY,
               CASE 
                   WHEN CDE_MC_REGION_CAP = 'CSOUT' THEN 'Southern'
                   WHEN CDE_MC_REGION_CAP = 'CCNTR' THEN 'Central'
                   WHEN CDE_MC_REGION_CAP = 'CNORT' THEN 'Northern'
                   WHEN CDE_MC_REGION_CAP = 'CGRBO' THEN 'Greater Boston'
                   WHEN CDE_MC_REGION_CAP = 'CWEST' THEN 'Western'
                   WHEN DSC_MC_REGION_CAP LIKE ' %' THEN 'Out of state/Unknown'
                   ELSE DSC_MC_REGION_CAP 
               END AS MCO_REGION,
               VALID_FROM_DT_TM, valid_thru_dt_tm
        FROM MHDWPROD.NW.nw_zip_cur
    ) zip
      ON zip.ADR_MAIL_ZIP = addr.ADR_ZIP_CODE 
      AND :today BETWEEN zip.VALID_FROM_DT_TM AND zip.valid_thru_dt_tm;


    /* Paste your existing CREATE OR REPLACE TABLE ACOEnr_02_EnrDis AS section here */

    /*
       IMPORTANT:
       Inside the ACOEnr_02_EnrDis query, replace all old variables:

       $StartDate  -> :StartDate
       $EndDate    -> :EndDate
       $today      -> :today
       $lastweek   -> :lastweek

       Example:
       and $EndDate between enr.VALID_FROM_DT_TM and enr.VALID_THRU_DT_TM

       becomes:

       and :EndDate between enr.VALID_FROM_DT_TM and enr.VALID_THRU_DT_TM
    */

    RETURN 'ACO Enrollment/Disenrollment weekly refresh completed for run date ' || :today;

END;
$$;



CREATE OR REPLACE TASK ACOENR_WEEKLY_REFRESH_TASK
    WAREHOUSE = MHA_WH
    SCHEDULE = 'USING CRON 0 9 * * WED America/New_York'
AS
    CALL ACOENR_WEEKLY_REFRESH();


    ALTER TASK ACOENR_WEEKLY_REFRESH_TASK RESUME;

    CALL ACOENR_WEEKLY_REFRESH();


    SELECT COUNT(*)
FROM ACOEnr_02_EnrDis;

SELECT
    MIN(startdt) AS min_dt,
    MAX(enddt)   AS max_dt,
    SUM(mbrs)    AS total_members
FROM ACOEnr_02_EnrDis;
